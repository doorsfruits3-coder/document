        function renderFileList(container, fileArray, emptyMessage) {
            if (!container) return;

            if (fileArray.length === 0) {
                container.innerHTML = `
                    

                        ${emptyMessage}
                    

                `;
                return;
            }

            let html = '';
            fileArray.forEach(file => {
                const icon = getIconForFile(file.name);
                const size = (file.size / 1024).toFixed(1);

                html += `
                    

                        

                            ${icon}
                            

                                
${file.name}

                                
${size} KB

                            

                        

                        
                    

                `;
            });
            container.innerHTML = html;
        }
